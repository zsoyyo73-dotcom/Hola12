local function IsRealPlayer(player)
    if not player or not player:IsA("Player") or player == LocalPlayer or player.UserId <= 0 then return false end
    
    local char = player.Character
    if not char or not char.Parent then return false end
    
    local hum = char:FindFirstChildOfClass("Humanoid")
    local head = char:FindFirstChild("Head")
    local root = char:FindFirstChild("HumanoidRootPart")
    
    if not hum or hum.Health <= 0 or not head or not root then return false end
    
    for _, part in pairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
            if part.Transparency + part.LocalTransparencyModifier < 0.9 then
                return true
            end
        end
    end
    
    return false
end

end

local function GetTextWidth(text, font, size)
    return TextService:GetTextSize(text, size, font, Vector2.new(1000, 1000)).X
end

local function GetActiveArrayListColor(index)
    if _G.Z_ArrayList_ColorMode == "RGB" then
        local hue = ((tick() * 0.35) + ((index or 1) * 0.06)) % 1
        return Color3.fromHSV(hue, 0.8, 1)
    elseif _G.Z_ArrayList_ColorMode == "Morado" then
        return Color3.fromRGB(170, 0, 255)
    elseif _G.Z_ArrayList_ColorMode == "Azul" then
        return Color3.fromRGB(0, 150, 255)
    elseif _G.Z_ArrayList_ColorMode == "Personalizado" then
        return _G.Z_ArrayList_CustomColor
    else
        return Color3.fromRGB(0, 255, 128)
    end
end

local function RenderArrayList()
    ActiveRenderItems = {}
    
    for _, child in pairs(arrayContainer:GetChildren()) do
        if child:IsA("Frame") then child:Destroy() end
    end

    if not _G.Z_ArrayList_Enabled then return end
    UpdateContainerLayout()

    local sortedList = {}
    for name, data in pairs(ActiveModules) do
        if data.State then
            table.insert(sortedList, {
                Name = name,
                Suffix = data.Suffix or ""
            })
        end
    end

    local font = Enum.Font.GothamBold
    local textSize = _G.Z_ArrayList_TextSize or 14

    table.sort(sortedList, function(a, b)
        local textA = a.Name .. (a.Suffix ~= "" and (" " .. a.Suffix) or "")
        local textB = b.Name .. (b.Suffix ~= "" and (" " .. b.Suffix) or "")
        return GetTextWidth(textA, font, textSize) > GetTextWidth(textB, font, textSize)
    end)

    for i, mod in ipairs(sortedList) do
        local fullText = mod.Name .. (mod.Suffix ~= "" and (" " .. mod.Suffix) or "")
        local textWidth = GetTextWidth(fullText, font, textSize)
        local currentColor = GetActiveArrayListColor(i)
        local itemHeight = textSize + 8

        local itemFrame = Instance.new("Frame", arrayContainer)
        itemFrame.Size = UDim2.new(0, textWidth + 18, 0, itemHeight)
        itemFrame.BackgroundColor3 = Color3.fromRGB(15, 15, 20)
        itemFrame.BackgroundTransparency = 0.25
        itemFrame.BorderSizePixel = 0
        itemFrame.LayoutOrder = i

        local itemCorner = Instance.new("UICorner", itemFrame)
        itemCorner.CornerRadius = UDim.new(0, 4)

        local itemStroke = Instance.new("UIStroke", itemFrame)
        itemStroke.Thickness = 1
        itemStroke.Transparency = 0.55
        itemStroke.Color = currentColor

        local accent = Instance.new("Frame", itemFrame)
        accent.Size = UDim2.new(0, 3, 1, -4)
        accent.Position = UDim2.new(1, -4, 0, 2)
        accent.BackgroundColor3 = currentColor
        accent.BorderSizePixel = 0

        local accentCorner = Instance.new("UICorner", accent)
        accentCorner.CornerRadius = UDim.new(0, 2)

        local label = Instance.new("TextLabel", itemFrame)
        label.Size = UDim2.new(1, -12, 1, 0)
        label.Position = UDim2.new(0, 4, 0, 0)
        label.BackgroundTransparency = 1
        label.Text = fullText
        label.TextColor3 = currentColor
        label.Font = font
        label.TextSize = textSize
        label.TextXAlignment = Enum.TextXAlignment.Right

        table.insert(ActiveRenderItems, {
            Index = i,
            Label = label,
            Accent = accent,
            Stroke = itemStroke
        })
    end
end

local function ExecuteKill()
    local target = getgenv().CurrentTarget
    if not target then return end
    
    local char = LocalPlayer.Character
    local backpack = LocalPlayer.Backpack
    local myRoot = char and char:FindFirstChild("HumanoidRootPart")
    
    local gun = char:FindFirstChild("Gun") or (backpack and backpack:FindFirstChild("Gun"))
    if not gun or not myRoot then return end

    -- Equipa automáticamente si no la tienes en mano
    if gun.Parent ~= char then 
        char.Humanoid:EquipTool(gun)
        task.wait(0.05)
    end

    -- Cálculo de predicción
    local rawPing = Stats.Network.ServerStatsItem["Data Ping"]:GetValueString():split(" ")[1]
    local pingValue = tonumber(rawPing) or 50
    local Prediction = (pingValue / 2500) + 0.175 
    local TargetPos = target.Position + (target.AssemblyLinearVelocity * Prediction)
    
    -- CORRECCIÓN: El origen debe salir de tu posición real para que el servidor no rechace el disparo
    local GunOrigin = myRoot.Position + Vector3.new(0, 2, 0)
    local args = { CFrame.new(GunOrigin), CFrame.new(TargetPos) }
    
    if gun:FindFirstChild("Shoot") then 
        gun.Shoot:FireServer(unpack(args))
        
        if V2_Active then
            task.wait(0.05)
            gun.Parent = backpack 
        end
    end
end

local function HackGun(v)
    if type(v) ~= "table" or not rawget(v, "gunName") then return end
    
    if _G.Z_GunMods_InfAmmo then
        local hugeAmmo = 999999999
        v.AmmoInGun = hugeAmmo 
        v.Ammo = hugeAmmo
        v.StoredAmmo = hugeAmmo
        v.MaxStoredAmmo = hugeAmmo 
        if rawget(v, "Jammed") ~= nil then v.Jammed = false end
    end

    if _G.Z_GunMods_NoRecoil then
        v.MinSpread = 0; v.MaxSpread = 0; v.AimInaccuracyStepAmount = 0; v.AimInaccuracyDecrease = 10
        v.MinRecoilPower = 0; v.MaxRecoilPower = 0; v.RecoilPowerStepAmount = 0; v.Recoil = 0 
        if type(v.camRecoil) == "table" then
            v.camRecoil.camRecoilUp = {0,0}; v.camRecoil.camRecoilTilt = {0,0}; v.camRecoil.camRecoilLeft = {0,0}; v.camRecoil.camRecoilRight = {0,0}
        end
        if type(v.gunRecoil) == "table" then
            v.gunRecoil.gunRecoilUp = {0,0}; v.gunRecoil.gunRecoilTilt = {0,0}; v.gunRecoil.gunRecoilLeft = {0,0}; v.gunRecoil.gunRecoilRight = {0,0}; v.gunRecoil.DownPunch = 0
        end
    end
end

task.spawn(function()
    local Camera = workspace.CurrentCamera
    local LP = Players.LocalPlayer

    if CoreGui:FindFirstChild("Z_TRACERS_LITE") then

    end

    local TracerGui = Instance.new("ScreenGui", CoreGui)
    TracerGui.Name = "Z_TRACERS_LITE"
    TracerGui.IgnoreGuiInset = true 

    local Tracer_Data = {}

    local function CreateTracer(player)
        if Tracer_Data[player] then return end
        local line = Instance.new("Frame")
        line.Name = "TracerLine"
        line.AnchorPoint = Vector2.new(0.5, 0.5) 
        line.BackgroundColor3 = _G.Z_Tracer_Color
        line.BackgroundTransparency = _G.Z_Tracer_Transparency
        line.BorderSizePixel = 0
        line.Visible = false
        line.Parent = TracerGui
        Tracer_Data[player] = line
    end

    local function RemoveTracer(player)
        if Tracer_Data[player] then
            pcall(function() Tracer_Data[player]:Destroy() end)
            Tracer_Data[player] = nil
        end
    end

    for _, p in pairs(Players:GetPlayers()) do if p ~= LP then CreateTracer(p) end end
    Players.PlayerAdded:Connect(function(p) if p ~= LP then CreateTracer(p) end end)
    Players.PlayerRemoving:Connect(RemoveTracer)

    RunService.RenderStepped:Connect(function()
        local ViewportSize = Camera.ViewportSize
        
        local Origin = Vector2.new(ViewportSize.X / 2, ViewportSize.Y)
        if _G.Z_Tracer_Origin == "Arriba" then
            Origin = Vector2.new(ViewportSize.X / 2, 0)
        elseif _G.Z_Tracer_Origin == "Centro" then
            Origin = Vector2.new(ViewportSize.X / 2, ViewportSize.Y / 2)
        end

        local safeWidth = math.max(1, _G.Z_Tracer_Width or 1)

        for player, line in pairs(Tracer_Data) do
            local char = player.Character
            local head = char and char:FindFirstChild("Head")
            local hum = char and char:FindFirstChildOfClass("Humanoid")

            -- Validación Anti-Bot idéntica a la del ESP 2X
            if not _G.Z_Tracers_Enabled or not head or not hum or hum.Health <= 0 or not IsRealPlayer(player) then
                line.Visible = false
                continue
            end

            if _G.Z_Tracer_TeamCheck and player.Team and LP.Team and player.Team == LP.Team then
                line.Visible = false
                continue
            end

            local headPos, onScreen = Camera:WorldToViewportPoint(head.Position)
            local Target

            if onScreen then
                Target = Vector2.new(headPos.X, headPos.Y)
            else
                local centerScreen = Vector2.new(ViewportSize.X / 2, ViewportSize.Y / 2)
                local fakePos = Vector2.new(headPos.X, headPos.Y)
                if headPos.Z < 0 then
                    fakePos = Vector2.new(
                        centerScreen.X + (centerScreen.X - headPos.X), 
                        centerScreen.Y + (centerScreen.Y - headPos.Y)
                    )
                end
                local direction = (fakePos - centerScreen).Unit
                Target = centerScreen + (direction * (ViewportSize.X * 0.5))
            end
            
            local Distance = (Target - Origin).Magnitude
            local Center = (Target + Origin) / 2
            local Angle = math.atan2(Target.Y - Origin.Y, Target.X - Origin.X)

            line.Size = UDim2.new(0, Distance, 0, safeWidth) 
            line.BackgroundTransparency = _G.Z_Tracer_Transparency or 0.4
            line.BackgroundColor3 = _G.Z_Tracer_Color
            line.Position = UDim2.new(0, Center.X, 0, Center.Y)
            line.Rotation = math.deg(Angle)
            line.Visible = true
        end
    end)

task.spawn(function()
    local RunService = game:GetService("RunService")
    local Players = game:GetService("Players")
    local Camera = workspace.CurrentCamera
    local LP = Players.LocalPlayer

    local RealCFrame = nil
    local CamAnchor = nil
    local WasDesyncing = false

    local function GetCamAnchor()
        if not CamAnchor then
            CamAnchor = Instance.new("Part")
            CamAnchor.Name = "Z_CamAnchor"
            CamAnchor.Transparency = 1
            CamAnchor.CanCollide = false
            CamAnchor.Anchored = true
            CamAnchor.Size = Vector3.new(1, 1, 1)
            CamAnchor.Parent = workspace
        end
        return CamAnchor
    end

    RunService.RenderStepped:Connect(function()
        local Char = LP.Character
        local HRP = Char and Char:FindFirstChild("HumanoidRootPart")
        local Hum = Char and Char:FindFirstChild("Humanoid")

        if _G.DesyncEnabled and HRP and Hum and RealCFrame then
            HRP.CFrame = RealCFrame
            
            local anchor = GetCamAnchor()
            anchor.Position = RealCFrame.Position + Vector3.new(0, 1.5, 0) 
            Camera.CameraSubject = anchor

            local camDist = (Camera.CFrame.Position - anchor.Position).Magnitude
            local isFirstPerson = camDist < 1.5

            for _, part in pairs(Char:GetDescendants()) do
                if part:IsA("BasePart") or part:IsA("Decal") then
                    if part.Name ~= "HumanoidRootPart" then
                        part.LocalTransparencyModifier = isFirstPerson and 1 or 0
                    end
                end
            end

        else
            if Hum and Camera.CameraSubject ~= Hum then
                Camera.CameraSubject = Hum
                
                if Char then
                    for _, part in pairs(Char:GetDescendants()) do
                        if part:IsA("BasePart") or part:IsA("Decal") then
                            part.LocalTransparencyModifier = 0
                        end
                    end
                end
            end
            if CamAnchor then
                CamAnchor:Destroy()
                CamAnchor = nil
            end
        end
    end)

    RunService.Heartbeat:Connect(function()
        local Char = LP.Character
        local HRP = Char and Char:FindFirstChild("HumanoidRootPart")
        
        if _G.DesyncEnabled and HRP then
            WasDesyncing = true
            
            RealCFrame = HRP.CFrame
            
            local rx = math.random(-_G.DesyncPower, _G.DesyncPower) / 10 
            local rz = math.random(-_G.DesyncPower, _G.DesyncPower) / 10 
            
            HRP.CFrame = RealCFrame * CFrame.new(rx, 0, rz) * CFrame.Angles(0, math.rad(math.random(0, 360)), 0)
            
        elseif WasDesyncing then
            if HRP and RealCFrame then
                HRP.CFrame = RealCFrame
                HRP.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                HRP.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
            end
            
            RealCFrame = nil
            WasDesyncing = false
        end
    end)
end)

_G.Z_Spin_Enabled = false

task.spawn(function()
    local Players = game:GetService("Players")
    local RunService = game:GetService("RunService")
    local LP = Players.LocalPlayer
    local spinAngle = 0

    RunService.Heartbeat:Connect(function()
        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local hum = char and char:FindFirstChildOfClass("Humanoid")

        if _G.Z_Spin_Enabled and hrp and hum and hum.Health > 0 then
            -- Desactivamos el auto-rotado para que la cámara y los disparos mantengan su dirección
            hum.AutoRotate = false

            -- Velocidad de giro fija determinada (45° por frame)
            spinAngle = (spinAngle + 45) % 360

            -- Giramos la CFrame sobre el eje Y respetando la posición actual
            hrp.CFrame = CFrame.new(hrp.CFrame.Position) * CFrame.Angles(0, math.rad(spinAngle), 0)
        else
            -- Al apagar el Spinbot, devolvemos el control de rotación normal al personaje
            if hum and not hum.AutoRotate then
                hum.AutoRotate = true
            end
        end
    end)
end)

_G.Z_Hitbox_Active = false
_G.Z_Hitbox_Size = 20
_G.Z_Hitbox_Mode = "Head"
_G.Z_Team_Check = true
_G.Z_Force_Reset = false

local s_p = "Si".."ze"
local h_p = "He".."ad"
local t_p = "Tor".."so"
local tr_p = "Trans".."parency"
local d_p = "Z".."Data".."99"

RunService.Heartbeat:Connect(function()
    if not _G.Z_Hitbox_Active and not _G.Z_Force_Reset then return end

    local lp = Players.LocalPlayer
    local val_size = _G.Z_Hitbox_Size or 20
    local v_size = Vector3.new(val_size, val_size, val_size)

    for _, p in pairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local char = p.Character

            local is_team = _G.Z_Team_Check and lp.Team and p.Team == lp.Team
            if not _G.Z_Hitbox_Active or _G.Z_Force_Reset or is_team then
                for _, part in pairs(char:GetChildren()) do
                    local backup = part:FindFirstChild(d_p)
                    if backup then
                        part[s_p] = backup.Value
                        part[tr_p] = 0
                        part.CanCollide = true
                    end
                end
                if is_team then continue end
            end

            if not _G.Z_Hitbox_Active then continue end

            local targets = {}
            if _G.Z_Hitbox_Mode == "Head" then
                local head = char:FindFirstChild(h_p) or char:FindFirstChild("FakeHead")
                if head then table.insert(targets, head) end
            elseif _G.Z_Hitbox_Mode == "Torso" then
                local torso = char:FindFirstChild(t_p) or char:FindFirstChild("Upper"..t_p) or char:FindFirstChild("Lower"..t_p)
                if torso then table.insert(targets, torso) end
            elseif _G.Z_Hitbox_Mode == "Huesos" then
                local bones = {"Left Arm", "Right Arm", "Left Leg", "Right Leg", "LeftUpperArm", "RightUpperArm", "LeftUpperLeg", "RightUpperLeg"}
                for _, b in pairs(bones) do 
                    local bp = char:FindFirstChild(b)
                    if bp then table.insert(targets, bp) end
                end
            end

            for _, obj in pairs(targets) do
                if obj and obj:IsA("BasePart") then
                    if obj[s_p] ~= v_size then
                        if not obj:FindFirstChild(d_p) then
                            local s = Instance.new("Vector3Value", obj)
                            s.Name = d_p
                            s.Value = obj[s_p]
                        end

                        obj[s_p] = v_size
                        obj[tr_p] = 0.6
                        obj.Color = Color3.fromRGB(255, 0, 0)
                        obj.Material = Enum.Material.Neon
                        obj.CanCollide = false
                        obj.Massless = true

                        local m = obj:FindFirstChildOfClass("SpecialMesh")
                        if m then m.Scale = Vector3.new(1,1,1) end 
                    end
                end
            end
        end
    end
    _G.Z_Force_Reset = false 
end)

_G.Aimbot_WallCheck = true

_G.Aimbot_Prediction = 0.12
_G.Aimbot_MaxDistance = 5000

local currentTarget = nil

local FOVCircle = Drawing.new("Circle")
FOVCircle.Thickness = 1.5
FOVCircle.Color = Color3.fromRGB(255, 50, 50)

local function getScreenCenter()
    return Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
end

local function isPartVisible(part)
    if not _G.Aimbot_WallCheck then return true end
    local char = LocalPlayer.Character
    if not char then return false end

    local origin = Camera.CFrame.Position
    local destination = part.Position
    local direction = destination - origin

    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.FilterDescendantsInstances = { char, part.Parent }
    raycastParams.IgnoreWater = true

    local result = workspace:Raycast(origin, direction, raycastParams)
    return result == nil
end

local function getTarget()
    local centerPos = getScreenCenter()
    local localChar = LocalPlayer.Character
    local localRoot = localChar and localChar:FindFirstChild("HumanoidRootPart")

    if currentTarget and currentTarget.Parent then
        local char = currentTarget.Parent
        local player = Players:GetPlayerFromCharacter(char)
        local humanoid = char:FindFirstChildOfClass("Humanoid")
        local rootPart = char:FindFirstChild("HumanoidRootPart")
        
        local isTeam = _G.Aimbot_TeamCheck and LocalPlayer.Team and player and player.Team == LocalPlayer.Team
        local passesAntiBot = (_G.IsRealPlayer and player) and _G.IsRealPlayer(player) or true

        local inRange = true
        if localRoot and rootPart then
            local worldDist = (rootPart.Position - localRoot.Position).Magnitude
            if worldDist > _G.Aimbot_MaxDistance then
                inRange = false
            end
        end

        if player and humanoid and humanoid.Health > 0 and inRange and passesAntiBot and not isTeam and isPartVisible(currentTarget) then
            local screenPos, onScreen = Camera:WorldToViewportPoint(currentTarget.Position)
            if onScreen then
                local dist = (Vector2.new(screenPos.X, screenPos.Y) - centerPos).Magnitude
                if dist <= _G.Aimbot_FOV then
                    return currentTarget
                end
            end
        end
    end

    currentTarget = nil
    local closest = nil
    local minDistance = _G.Aimbot_FOV

    local validPlayers = _G.GetRealPlayers and _G.GetRealPlayers() or Players:GetPlayers()

    for _, player in pairs(validPlayers) do
        if player ~= LocalPlayer and player.Character then
            local isTeam = _G.Aimbot_TeamCheck and LocalPlayer.Team and player.Team == LocalPlayer.Team
            if not isTeam then
                local character = player.Character
                local targetPart = character:FindFirstChild(_G.Aimbot_Part)
                local humanoid = character:FindFirstChildOfClass("Humanoid")
                local rootPart = character:FindFirstChild("HumanoidRootPart")

                if targetPart and humanoid and humanoid.Health > 0 and rootPart then
                    local worldDist = localRoot and (rootPart.Position - localRoot.Position).Magnitude or 0
                    if worldDist <= _G.Aimbot_MaxDistance then
                        local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                        if onScreen and isPartVisible(targetPart) then
                            local dist = (Vector2.new(screenPos.X, screenPos.Y) - centerPos).Magnitude
                            if dist < minDistance then
                                minDistance = dist
                                closest = targetPart
                            end
                        end
                    end
                end
            end
        end
    end

    currentTarget = closest
    return currentTarget
end

RunService.RenderStepped:Connect(function()
    local centerPos = getScreenCenter()
    FOVCircle.Position = centerPos
    FOVCircle.Radius = _G.Aimbot_FOV
    FOVCircle.Visible = _G.Aimbot_ShowFOV and _G.Aimbot_Enabled

    if _G.Aimbot_Enabled then
        local target = getTarget()
        if target then
            local char = target.Parent
            local rootPart = char and char:FindFirstChild("HumanoidRootPart")
            local velocity = (rootPart and rootPart.AssemblyLinearVelocity) or Vector3.new(0, 0, 0)
            local predictedPosition = target.Position + (velocity * _G.Aimbot_Prediction)
            
            local targetCFrame = CFrame.new(Camera.CFrame.Position, predictedPosition)
            Camera.CFrame = Camera.CFrame:Lerp(targetCFrame, _G.Aimbot_Smoothness)
        end
    else
        currentTarget = nil
    end
end)

local function getClosestTargetSilent()
    local centerPos = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local char = LocalPlayer.Character
    if not char or not char.PrimaryPart then return nil end
    local localPos = char.PrimaryPart.Position
    local closest = nil
    local minDist = math.huge

    local validPlayers = _G.GetRealPlayers and _G.GetRealPlayers() or Players:GetPlayers()

    for _, pl in pairs(validPlayers) do
        if pl ~= LocalPlayer and pl.Character and pl.Character:FindFirstChild("Head") then
            local isTeam = _G.Silent_TeamCheck and LocalPlayer.Team and pl.Team and pl.Team == LocalPlayer.Team
            if not isTeam then
                local humanoid = pl.Character:FindFirstChildOfClass("Humanoid")
                if humanoid and humanoid.Health > 0 then
                    local passesAntiBot = _G.IsRealPlayer and _G.IsRealPlayer(pl) or true
                    if passesAntiBot then
                        local head = pl.Character.Head
                        local dist = (localPos - head.Position).Magnitude
                        
                        if dist <= _G.Silent_Range then
                            local sp, onScreen = Camera:WorldToViewportPoint(head.Position)
                            if onScreen then
                                local screenDist = (Vector2.new(sp.X, sp.Y) - centerPos).Magnitude
                                if screenDist <= _G.Silent_FOV and dist < minDist then
                                    minDist = dist
                                    closest = head
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    return closest
end

local function checkWeapon()
    local char = LocalPlayer.Character

        WeaponEquipped = false
        return

    local tool = char:FindFirstChildWhichIsA("Tool")
    WeaponEquipped = tool and true or false
end

local oldNamecall
oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(...)
    local method = getnamecallmethod()
    local args = {...}
    local self = args[1]

    if _G.Silent_Enabled and not checkcaller() and self == workspace and method == "Raycast" then
        local origin = args[2]
        local direction = args[3]
        local params = args[4]
        if typeof(origin) == "Vector3" and typeof(direction) == "Vector3" and CurrentTargetSilent then
            if not params then
                params = RaycastParams.new()
            end
            pcall(function()
                params.FilterType = Enum.RaycastFilterType.Include
                params.FilterDescendantsInstances = {CurrentTargetSilent.Parent}
            end)
            args[3] = (CurrentTargetSilent.Position - origin).Unit * math.max(direction.Magnitude, (CurrentTargetSilent.Position - origin).Magnitude + 500)
            args[4] = params
            return oldNamecall(unpack(args))
        end
    end

    if _G.Silent_Enabled and not checkcaller() and WeaponEquipped then
        local recoilMethods = {"ApplyRecoil", "AddRecoil", "DoRecoil"}
        for _, rm in pairs(recoilMethods) do
            if method == rm then return nil end
        end
    end

    return oldNamecall(...)
end))

local ConfigLimit = 5

local function GetConfigFolderPath()
    local path1 = "WindUI/" .. Window.Folder .. "/config"
    local path2 = "WindUI/" .. Window.Folder .. "/configs"
    if isfolder and isfolder(path1) then return path1 end
    if isfolder and isfolder(path2) then return path2 end
    return path1
end

local function GetSavedConfigs()
    local list = {}
    local path = GetConfigFolderPath()
    if listfiles and isfolder and isfolder(path) then
        local files = listfiles(path)
        for _, filePath in ipairs(files) do
            local fileName = filePath:match("([^/\\]+)%.json$")
            if fileName and fileName ~= "" then
                table.insert(list, fileName)
            end
        end
    end
    return list
end

local StatusParagraph = ConfigTab:Paragraph({
    Title = "Estado de Configuraciones",
    Desc = "Cargando..."
})

local function RefreshConfigList()
    local saved = GetSavedConfigs()
    StatusParagraph:SetDesc("Configuraciones guardadas: " .. tostring(#saved) .. " / " .. tostring(ConfigLimit))
    if ConfigDropdown then
        if ConfigDropdown.SetValues then
            ConfigDropdown:SetValues(saved)
        elseif ConfigDropdown.Refresh then
            ConfigDropdown:Refresh(saved)
        end
    end
end
